Update element in all perticuler space(with where clause)
 --> db.employee.updateMany({"skills" :"Accounting"},{$set:{"skills.$" :"ACC"}})


Add new Element in perticuler space
 --> db.employee.updateMany({_id:6},{$push:{"skills" :"GST"}})

 Add new Element in All records
 --> db.employee.updateMany({},{$push:{"skills":{$each:["GST","Cricket"]}}})


 Add new element where does not exist
 -->db.employee.updateMany({},{$addToSet:{"skills":"ACC"}})

 -->db.employee.updateMany({},{$pop:{"skills":1}})

 Remove first element
 -->db.employee.updateMany({},{$pop:{"skills":-1}})
